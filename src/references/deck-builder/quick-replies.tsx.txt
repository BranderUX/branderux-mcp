import { useState } from "react";
import { Box, ButtonBase, Typography } from "@mui/material";

type Reply = { id?: string; label: string; query: string };
type WhatsApp = { label?: string; text?: string; number?: string; from?: string; note?: string };
type Chip = { id: string; label: string; query: string };

export interface Props {
  replies: Reply[];
  heading?: string;
  whatsapp?: WhatsApp;
  onAsk?: (chip: Chip) => void;
}

const NAVY = "#17324D";
const MUTED = "#55626F";
const WA_GREEN = "#25D366";
const WA_INK = "#062A17";
const FOCUS = { fontFamily: "inherit", "&:focus-visible": { outline: "3px solid", outlineColor: NAVY, outlineOffset: 2 } };
const REF_CHARS = "ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const newRef = () => Array.from({ length: 4 }, () => REF_CHARS[Math.floor(Math.random() * REF_CHARS.length)]).join("");

const WA_GLYPH =
  "M17.472 14.382c-.297-.149-1.758-.867-2.03-.967-.273-.099-.471-.148-.67.15-.197.297-.767.966-.94 1.164-.173.199-.347.223-.644.075-.297-.15-1.255-.463-2.39-1.475-.883-.788-1.48-1.761-1.653-2.059-.173-.297-.018-.458.13-.606.134-.133.298-.347.446-.52.149-.174.198-.298.298-.497.099-.198.05-.371-.025-.52-.075-.149-.669-1.612-.916-2.207-.242-.579-.487-.5-.669-.51-.173-.008-.371-.01-.57-.01-.198 0-.52.074-.792.372-.272.297-1.04 1.016-1.04 2.479 0 1.462 1.065 2.875 1.213 3.074.149.198 2.096 3.2 5.077 4.487.709.306 1.262.489 1.694.625.712.227 1.36.195 1.871.118.571-.085 1.758-.719 2.006-1.413.248-.694.248-1.289.173-1.413-.074-.124-.272-.198-.57-.347m-5.421 7.403h-.004a9.87 9.87 0 0 1-5.031-1.378l-.361-.214-3.741.982.998-3.648-.235-.374a9.86 9.86 0 0 1-1.51-5.26c.001-5.45 4.436-9.884 9.888-9.884 2.64 0 5.122 1.03 6.988 2.898a9.825 9.825 0 0 1 2.893 6.994c-.003 5.45-4.437 9.884-9.885 9.884m8.413-18.297A11.815 11.815 0 0 0 12.050 0C5.495 0 .16 5.335.157 11.892c0 2.096.547 4.142 1.588 5.945L.057 24l6.305-1.654a11.882 11.882 0 0 0 5.683 1.448h.005c6.554 0 11.89-5.335 11.893-11.893a11.821 11.821 0 0 0-3.48-8.413Z";

function WhatsAppGlyph({ size }: { size: number }) {
  return (
    <svg width={size} height={size} viewBox="0 0 24 24" aria-hidden="true" focusable="false" style={{ display: "block", flexShrink: 0 }}>
      <path fill="#FFFFFF" d={WA_GLYPH} />
    </svg>
  );
}

export default function Component({ replies, heading, whatsapp, onAsk }: Props) {
  const [ref] = useState(newRef);
  const [picked, setPicked] = useState("");
  const list = (replies || []).filter((reply) => reply && reply.label && reply.query).slice(0, 5);
  const waNumber = String(whatsapp?.number || "15550106060").replace(/\D/g, "");
  const waMessage = `${whatsapp?.text || "Hi Brandeck, I have a question about a deck."} (ref ${ref})`;
  const waHref = `https://wa.me/${waNumber}?text=${encodeURIComponent(waMessage)}`;

  return (
    <Box sx={{ width: "100%", maxWidth: 680, marginInline: "auto", display: "flex", flexDirection: "column", alignItems: "flex-end", gap: 1 }}>
      {heading ? <Typography sx={{ fontSize: 13, color: MUTED, textAlign: "end" }}>{heading}</Typography> : null}
      <Box role="group" aria-label={heading || "Quick replies"} sx={{ display: "flex", flexWrap: "wrap", justifyContent: "flex-end", gap: 1 }}>
        {list.map((reply) => {
          const on = picked === reply.query;
          return (
            <ButtonBase
              key={reply.id || reply.query}
              aria-pressed={on}
              onClick={() => {
                setPicked(reply.query);
                onAsk?.({ id: reply.id || reply.query, label: reply.label, query: reply.query });
              }}
              sx={{ px: 2, py: 1.15, minHeight: 44, borderRadius: "999px", border: "1.5px solid", borderColor: NAVY, bgcolor: on ? NAVY : "#FFFFFF", color: on ? "#FFFFFF" : NAVY, fontSize: 15, fontWeight: 600, lineHeight: 1.25, textAlign: "center", "&:hover": { bgcolor: on ? NAVY : "rgba(23,50,77,0.06)" }, ...FOCUS }}
            >
              {reply.label}
            </ButtonBase>
          );
        })}
      </Box>
      {whatsapp ? (
        <Box sx={{ display: "flex", flexDirection: "column", alignItems: "flex-end", gap: 0.5, mt: 0.5 }}>
          <ButtonBase
            component="a"
            href={waHref}
            target="_blank"
            rel="noopener noreferrer"
            onClick={() => onAsk?.({ id: "whatsapp", label: whatsapp.label || "Send it to me on WhatsApp", query: `Opened WhatsApp from ${whatsapp.from || "the chat"} (ref ${ref})` })}
            sx={{ px: 2, py: 1.15, minHeight: 44, borderRadius: "999px", bgcolor: WA_GREEN, color: WA_INK, fontSize: 15, fontWeight: 700, display: "flex", alignItems: "center", gap: 0.9, textDecoration: "none", "&:hover": { bgcolor: "#3BDB77" }, ...FOCUS }}
          >
            <WhatsAppGlyph size={20} />
            {whatsapp.label || "Send it to me on WhatsApp"}
          </ButtonBase>
          {whatsapp.note ? <Typography sx={{ fontSize: 12.5, color: MUTED, textAlign: "end" }}>{whatsapp.note}</Typography> : null}
        </Box>
      ) : null}
    </Box>
  );
}
